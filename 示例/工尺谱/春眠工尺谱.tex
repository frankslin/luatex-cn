% 工尺谱示例 — 孟浩然《春晓》配工尺谱
% 演示 luatex-cn 的 \工尺 与 \音 命令
%
% API 速查：
%   \工尺{主字}{右侧工尺串}                    主字 + 工尺旁注
%   \工尺[缩放=0.6, 颜色=red]{主}{尺工}        参数化
%   \音{尺}                                     单独工尺字 (无板眼)
%   \音[板]{尺}                                 工尺字 + 红色板点
%   \音[眼]{尺}                                 工尺字 + 红色眼点
%   \音[头板/底板/底叮]{尺}                     其他常见板眼标记
%   \工尺设置{颜色=black, 缩放=0.55}             全局默认
%
\documentclass{ltc-guji}

\title{春\空格 晓}

\chapter{孟浩然·工尺谱}

\begin{document}
\begin{正文}

% 全局默认：工尺字略大、黑色
\工尺设置{缩放=0.55}

% ------------------------------------------------------------------
% 第一句 — 春眠不觉晓
% ------------------------------------------------------------------
\工尺{春}{\音[板]{尺}\音[眼]{工}}%
\工尺{眠}{\音[板]{六}}%
\工尺{不}{\音[眼]{五}\音[板]{尺}}%
\工尺{觉}{\音[眼]{工}}%
\工尺{晓}{\音[板]{六}\音[眼]{五}}

% ------------------------------------------------------------------
% 第二句 — 处处闻啼鸟
% ------------------------------------------------------------------
\工尺{处}{\音[板]{尺}}%
\工尺{处}{\音[眼]{工}}%
\工尺{闻}{\音[板]{六}\音[眼]{五}}%
\工尺{啼}{\音[板]{尺}\音[眼]{工}}%
\工尺{鸟}{\音[板]{六}\音[眼]{五}\音[板]{四}}

% ------------------------------------------------------------------
% 第三句 — 夜来风雨声
% ------------------------------------------------------------------
\工尺{夜}{\音[板]{尺}\音[眼]{工}\音[板]{六}}%
\工尺{来}{\音[眼]{五}\音[板]{尺}}%
\工尺{风}{\音[眼]{工}}%
\工尺{雨}{\音[板]{六}\音[眼]{五}}%
\工尺{声}{\音[板]{尺}}

% ------------------------------------------------------------------
% 第四句 — 花落知多少
% ------------------------------------------------------------------
\工尺{花}{\音[眼]{工}}%
\工尺{落}{\音[板]{六}}%
\工尺{知}{\音[眼]{五}}%
\工尺{多}{\音[板]{尺}}%
\工尺{少}{\音[眼]{工}\音[板]{六}\音[眼]{五}}

\end{正文}
\end{document}
